//! Scratch (review 2026-11-10, not for landing): play a mould dish from a
//! written hand, one action a line, and write the window's own frames.
//!
//! ```text
//! cargo run --release --example mouldhand -- seed=1 script=hand.txt out=DIR shots=every5 gif=20-40
//! ```
//! Script lines (`#` comments), times in seconds, positions in dish cells:
//! `T lamp X Y DUR` | `T sweep X1 Y1 X2 Y2 DUR` | `T oats X Y` |
//! `T salt X1 Y1 X2 Y2` | `T snip X1 Y1 X2 Y2`

use pixel_physics::mould::level::{DISHES, STRAY_REACH};
use pixel_physics::mould::look::Style;
use pixel_physics::mould::render::{self, View};
use pixel_physics::mould::{tune, Dish, Params, Phase, Tool, STEPS_PER_FRAME, STEPS_PER_SECOND};

#[derive(Clone, Debug)]
enum Act {
    Lamp { a: (f32, f32), b: (f32, f32), dur: f32 },
    Oats(f32, f32),
    Stroke(Tool, (f32, f32), (f32, f32)),
}

fn main() {
    let mut seed = 1u64;
    let mut level: Option<usize> = None;
    let mut script = String::new();
    let mut out = String::from(".");
    let mut shots_every = 10u64;
    let mut shot_list: Vec<u64> = Vec::new();
    let mut gif: Option<(u64, u64)> = None;
    let mut gif_every = 6u32;
    let mut scale = 3u32;
    let mut style = Style::GAME;
    let mut params = Params::default();
    let mut tag = String::from("hand");
    let mut stop: Option<u64> = None;
    let mut cursor_on = true;
    let mut show_walkers = false;
    let mut ghost = false;
    for arg in std::env::args().skip(1) {
        let (k, v) = arg.split_once('=').unwrap_or_else(|| panic!("bad arg {arg}"));
        match k {
            "seed" => seed = v.parse().unwrap(),
            "level" => level = Some(v.parse::<usize>().unwrap() - 1),
            "script" => script = std::fs::read_to_string(v).unwrap(),
            "out" => out = v.into(),
            "tag" => tag = v.into(),
            "stop" => stop = Some(v.parse().unwrap()),
            "cursor" => cursor_on = v == "1",
            "walkers" => show_walkers = v == "1",
            "ghost" => ghost = v == "1",
            "shots" => {
                if let Some(n) = v.strip_prefix("every") {
                    shots_every = n.parse().unwrap();
                } else {
                    shots_every = 0;
                    shot_list = v.split(',').map(|s| s.parse().unwrap()).collect();
                }
            }
            "gif" => {
                let (a, b) = v.split_once('-').unwrap();
                gif = Some((a.parse().unwrap(), b.parse().unwrap()));
            }
            "gifevery" => gif_every = v.parse().unwrap(),
            "scale" => scale = v.parse().unwrap(),
            "style" => style = Style::parse(v).unwrap(),
            "sensor_angle" => params.sensor_angle = v.parse::<f32>().unwrap().to_radians(),
            "spared" => params.spared = v.parse().unwrap(),
            "spared_scent" => params.spared_scent = v.parse().unwrap(),
            "settle_seconds" => params.settle_seconds = v.parse().unwrap(),
            "sensor_offset" => params.sensor_offset = v.parse().unwrap(),
            "food_deposit" => params.food_deposit = v.parse().unwrap(),
            "flow" => params.flow = v == "1",
            "grey" => {
                pixel_physics::mould::look::STARVE_LOOK.store(v.parse().unwrap(), std::sync::atomic::Ordering::Relaxed);
            }
            "tune" => {
                let bad = tune::from_text(&std::fs::read_to_string(v).unwrap(), &mut params);
                assert!(bad.is_empty(), "{bad:?}");
            }
            _ => panic!("unknown {k}"),
        }
    }
    println!("mouldhand seed={seed} level={level:?} tag={tag} scale={scale} sensor_angle={:.1} flow={} spared={}", params.sensor_angle.to_degrees(), params.flow, params.spared);
    let mut acts: Vec<(f32, Act)> = Vec::new();
    for line in script.lines() {
        let line = line.split('#').next().unwrap().trim();
        if line.is_empty() {
            continue;
        }
        let w: Vec<&str> = line.split_whitespace().collect();
        let f = |i: usize| w[i].parse::<f32>().unwrap_or_else(|_| panic!("bad line {line}"));
        let t = f(0);
        let act = match w[1] {
            "lamp" => Act::Lamp { a: (f(2), f(3)), b: (f(2), f(3)), dur: f(4) },
            "sweep" => Act::Lamp { a: (f(2), f(3)), b: (f(4), f(5)), dur: f(6) },
            "oats" => Act::Oats(f(2), f(3)),
            "salt" => Act::Stroke(Tool::Salt, (f(2), f(3)), (f(4), f(5))),
            "snip" => Act::Stroke(Tool::Snip, (f(2), f(3)), (f(4), f(5))),
            other => panic!("unknown act {other}"),
        };
        acts.push((t, act));
    }
    acts.sort_by(|a, b| a.0.total_cmp(&b.0));
    println!("{} actions", acts.len());

    let mut d = match level {
        Some(k) => Dish::from_level_with(&DISHES[k], seed, Params { stray_reach: STRAY_REACH, ..params.clone() }),
        None => Dish::with_params(seed, params.clone()),
    };
    println!("piles at start: {:?}", d.piles.iter().map(|p| (p.x as i32, p.y as i32, p.oats)).collect::<Vec<_>>());
    println!("pending: {:?}", d.pending.iter().map(|(t, p)| (t / STEPS_PER_SECOND as u64, p.x as i32, p.y as i32)).collect::<Vec<_>>());
    println!("spared: {:?}", d.spared);
    let sps = STEPS_PER_SECOND as u64;
    let mut next = 0usize;
    let mut lamp: Option<(f32, Act)> = None;
    let mut frames = Vec::new();
    let mut frame = 0u32;
    let mut view = View { keys: false, style, ..View::default() };
    let mut over_at: Option<u64> = None;
    let mut ever = vec![0.0f32; d.trail.len()];
    let end_free = Dish::dish_steps() + Dish::storm_steps() + 3 * sps;
    loop {
        let secs = d.t as f32 / sps as f32;
        // actions due
        while next < acts.len() && acts[next].0 <= secs {
            let (t, act) = acts[next].clone();
            next += 1;
            match act {
                Act::Lamp { .. } => {
                    view.tool = Tool::Lamp;
                    lamp = Some((t, act));
                }
                Act::Oats(x, y) => {
                    view.tool = Tool::Oats;
                    view.cursor = Some((x, y));
                    let ok = d.place_oats(x, y);
                    println!("  {secs:6.1}s oats at {x},{y}: {}", if ok { "placed" } else { "REFUSED" });
                }
                Act::Stroke(tool, a, b) => {
                    view.tool = tool;
                    view.cursor = Some(b);
                    let ok = d.begin_stroke(tool, a.0, a.1);
                    if ok {
                        for k in 1..=20 {
                            let u = k as f32 / 20.0;
                            d.drag_stroke(a.0 + (b.0 - a.0) * u, a.1 + (b.1 - a.1) * u);
                        }
                        d.end_stroke();
                    }
                    println!("  {secs:6.1}s {} {a:?}->{b:?}: {}", tool.name(), if ok { "done" } else { "REFUSED" });
                }
            }
        }
        match &lamp {
            Some((t0, Act::Lamp { a, b, dur })) if secs < t0 + dur => {
                let u = ((secs - t0) / dur).clamp(0.0, 1.0);
                let p = (a.0 + (b.0 - a.0) * u, a.1 + (b.1 - a.1) * u);
                d.set_lamp(Some(p));
                view.cursor = Some(p);
            }
            _ => {
                d.set_lamp(None);
                lamp = None;
            }
        }
        if !cursor_on {
            view.cursor = None;
        }
        for _ in 0..STEPS_PER_FRAME {
            d.step();
        }
        if ghost {
            // Mock for the review: where tube has ever been, fading slowly.
            for i in 0..d.trail.len() {
                let now = if d.trail[i] > d.params.tube { 1.0 } else { 0.0 };
                ever[i] = (ever[i] * 0.9997f32).max(now);
            }
        }
        let s = d.t / sps;
        let on_second = d.t.is_multiple_of(sps);
        let want_shot = on_second && ((shots_every > 0 && s.is_multiple_of(shots_every)) || shot_list.contains(&s));
        if want_shot || gif.is_some_and(|(a, b)| s >= a && s < b && frame.is_multiple_of(gif_every)) {
            let (w, h) = render::frame_size(scale);
            let mut buf = vec![0u8; (w * h * 4) as usize];
            render::draw(&d, &view, scale, &mut buf);
            let mut img = image::RgbaImage::from_raw(w, h, buf).unwrap();
            if ghost {
                for i in 0..d.trail.len() {
                    if ever[i] > 0.02 && d.trail[i] <= d.params.tube * 0.6 {
                        let (cx, cy) = ((i % 256) as u32, (i / 256) as u32);
                        for oy in 0..scale {
                            for ox in 0..scale {
                                let (qx, qy) = (cx * scale + ox, cy * scale + oy);
                                if qx < w && qy < h {
                                    let p = img.get_pixel_mut(qx, qy);
                                    let e = ever[i];
                                    p.0[0] = p.0[0].saturating_add((34.0 * e) as u8);
                                    p.0[1] = p.0[1].saturating_add((40.0 * e) as u8);
                                    p.0[2] = p.0[2].saturating_add((22.0 * e) as u8);
                                }
                            }
                        }
                    }
                }
            }
            if show_walkers {
                // Mock for the review: the walkers themselves, as specks.
                for wk in &d.walkers {
                    let (px, py) = ((wk.x * scale as f32) as u32, (wk.y * scale as f32) as u32);
                    for oy in 0..scale / 2 {
                        for ox in 0..scale / 2 {
                            let (qx, qy) = (px + ox + scale / 4, py + oy + scale / 4);
                            if qx < w && qy < h {
                                let p = img.get_pixel_mut(qx, qy);
                                p.0[0] = p.0[0].saturating_add(70);
                                p.0[1] = p.0[1].saturating_add(60);
                                p.0[2] = p.0[2].saturating_add(30);
                            }
                        }
                    }
                }
            }
            if want_shot {
                img.save(format!("{out}/{tag}-{s:03}.png")).unwrap();
            }
            if gif.is_some_and(|(a, b)| s >= a && s < b && frame.is_multiple_of(gif_every)) {
                frames.push(img);
            }
        }
        if on_second && s.is_multiple_of(5) {
            let m = d.measure();
            let grey = (0..d.trail.len()).filter(|&i| d.trail[i] > d.params.tube && d.fed[i] == u16::MAX).count();
            let tube = (0..d.trail.len()).filter(|&i| d.trail[i] > d.params.tube).count();
            println!(
                "t={s:3}s walkers {:5} joined {}/{} ratio {:.2} tube {tube} unfed-tube {grey} oats {} salt {} snips {} on_patch {:?}",
                d.walkers.len(),
                m.reach,
                m.piles,
                m.ratio(),
                d.oats_left,
                d.salt_left,
                d.snips_left,
                d.on_patch()
            );
        }
        frame += 1;
        if d.phase == Phase::Over && over_at.is_none() {
            over_at = Some(d.t);
        }
        if stop.is_some_and(|x| s >= x) {
            break;
        }
        match (level, over_at) {
            (Some(_), Some(t)) if d.t >= t + 2 * sps => break,
            (None, _) if d.t >= end_free => break,
            _ => {}
        }
        if d.t > 600 * sps {
            break;
        }
    }
    if let Some(s) = &d.score {
        println!("SCORE stars {:?} reach {}/{} after {} ratio {:.2} joined_at {:?} on_patch {:?}", s.stars, s.reach, s.piles, s.after, s.ratio(), s.joined_at, s.on_patch);
    }
    if let Some(c) = &d.card {
        println!("CARD stars {} | {} | {}", c.stars, c.amount, c.why);
    }
    let c = &d.counts;
    println!("COUNTS births {} starved {} hungry {} snipped {} stormed {} oats {} snips {} joins {}", c.births, c.starved, c.hungry, c.snipped, c.stormed, c.oats_placed, c.snips_used, c.snaps);
    if !frames.is_empty() {
        let delay = image::Delay::from_numer_denom_ms(gif_every * 1000 / 60, 1);
        let n = frames.len();
        let fr: Vec<image::Frame> = frames.into_iter().map(|img| image::Frame::from_parts(img, 0, 0, delay)).collect();
        let file = std::fs::File::create(format!("{out}/{tag}.gif")).unwrap();
        let mut enc = image::codecs::gif::GifEncoder::new_with_speed(file, 10);
        enc.set_repeat(image::codecs::gif::Repeat::Infinite).unwrap();
        enc.encode_frames(fr).unwrap();
        println!("gif {out}/{tag}.gif {n} frames");
    }
}
